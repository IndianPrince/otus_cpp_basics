cmake_minimum_required(VERSION 3.14)

project(containers CXX)

set(CMAKE_CXX_STANDARD 17)

# Fetch GoogleTest at specific revision
include(FetchContent)
FetchContent_Declare(
    googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG 6910c9d
)
# For MSVC: use shared runtime to avoid conflicts in the IDE
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

set(EXECUTABLE_OUTPUT_PATH ${CMAKE_BINARY_DIR})
set(LIBRARY_OUTPUT_PATH ${CMAKE_BINARY_DIR})

include_directories(
    ${PROJECT_SOURCE_DIR}/inc
    ${PROJECT_SOURCE_DIR}/src
    )

file(GLOB all_SRCS
    "${PROJECT_SOURCE_DIR}/inc/*.h"
    "${PROJECT_SOURCE_DIR}/src/*.cpp"
)

# set_target_properties(hello_world PROPERTIES
#     CXX_STANDARD 17
#     CXX_STANDARD_REQUIRED ON
# )

add_executable(containers ${all_SRCS})

add_executable(
    test_containers
    test/carray_test.cpp
    test/list_test.cpp
    test/container_test_funcs.h)

target_include_directories( test_containers PRIVATE 
                            "${CMAKE_CURRENT_SOURCE_DIR}/inc"
                            "${CMAKE_CURRENT_SOURCE_DIR}/src"
                            )

target_link_libraries(test_containers PRIVATE gtest)

include(CTest)
if (BUILD_TESTING)
  add_test(NAME containers_tests COMMAND test_containers)
endif()
